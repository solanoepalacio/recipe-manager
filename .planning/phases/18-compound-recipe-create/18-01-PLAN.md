Add `BadRequestException` to the `@nestjs/common` import in `recipes.service.ts`.

Replace the `create` method body with:
```typescript
async create(userId: string, householdId: string, dto: CreateRecipeDto): Promise<RecipeDetailResponse> {
  const slug = await this.generateUniqueSlug(dto.name, householdId);

  try {
    const recipe = await this.prisma.$transaction(async (tx: any) => {
      return tx.recipe.create({
        data: {
          householdId,
          createdById: userId,
          name: dto.name,
          slug,
          description: dto.description,
          servingsQty: dto.servingsQty,
          servingsUnit: dto.servingsUnit,
          prepTime: dto.prepTime,
          cookTime: dto.cookTime,
          totalTime: dto.totalTime,
          performTime: dto.performTime,
          sourceUrl: dto.sourceUrl,
          sections: {
            create: [{
              title: null,
              order: 0,
              ...(dto.ingredients?.length
                ? {
                    ingredients: {
                      create: dto.ingredients.map((item, i) => ({
                        foodId: item.foodId,
                        unitId: item.unitId ?? null,
                        quantity: item.quantity ?? null,
                        note: item.note ?? null,
                        order: i,
                      })),
                    },
                  }
                : {}),
            }],
          },
          ...(dto.steps?.length
            ? {
                steps: {
                  create: dto.steps.map((step, i) => ({
                    title: step.title ?? null,
                    body: step.body,
                    order: i,
                  })),
                },
              }
            : {}),
        },
        include: RECIPE_INCLUDE,
      });
    });
    return toRecipeDetailResponse(recipe);
  } catch (error) {
    if (
      error instanceof Prisma.PrismaClientKnownRequestError &&
      error.code === 'P2003'
    ) {
      throw new BadRequestException('Invalid ingredient data: food or unit not found');
    }
    throw error;
  }
}
```

Key points:
- `generateUniqueSlug` stays OUTSIDE the `$transaction` (read-only, no writes)
- Always use `$transaction` (single code path per locked decision)
- Use `dto.ingredients?.length` (not just `dto.ingredients`) to handle both undefined and empty array
- `BadRequestException` (not `NotFoundException`) for P2003 per locked decision
- Error message is exactly `'Invalid ingredient data: food or unit not found'`
